export function parseArgs(argv, env = process.env, cwd = process.cwd()) {
  const flags = new Set()
  const positional = []
  let root = env.AI_PROJECT_ROOT || cwd
  let rootGiven = Boolean(env.AI_PROJECT_ROOT)

  for (let i = 0; i < argv.length; i += 1) {
    const arg = argv[i]
    if (arg === '--root') {
      const value = argv[i + 1]
      if (!value) throw new Error('--root требует путь к корню проекта')
      root = value
      rootGiven = true
      i += 1
    } else if (arg.startsWith('--root=')) {
      const value = arg.slice('--root='.length)
      if (!value) throw new Error('--root требует путь к корню проекта')
      root = value
      rootGiven = true
    } else if (arg.startsWith('--')) {
      flags.add(arg)
    } else {
      positional.push(arg)
    }
  }

  return { root: path.resolve(root), rootGiven, flags, positional }
}

// Мержит, а не заменяет: instructions, чужие провайдеры, агенты и прочие
// ключи проекта остаются на месте. Меняются только model/default_agent и
// описания моделей Ollama (нужны для облачных тегов: их нет в автодискавери).
export function mergeOpencodeConfig(profile, existing) {
  const base = baseOpencodeConfig()
  const cfg = existing && typeof existing === 'object' ? structuredClone(existing) : {}

  cfg.$schema = cfg.$schema || base.$schema

  const providers = { ...(cfg.providers || {}) }
  const ollama = { ...(providers.ollama || {}) }
  ollama.models = { ...base.providers.ollama.models, ...(ollama.models || {}) }
  providers.ollama = ollama
  cfg.providers = providers

  const mergedAgents = { ...base.agents, ...(cfg.agents || {}) }
  cfg.agents = Object.fromEntries(
    Object.entries(mergedAgents).map(([name, value]) => [
      name,
      { ...(base.agents[name] || {}), ...(value || {}) },
    ]),
  )

  cfg.model = profile.opencode.model
  cfg.default_agent = profile.opencode.agent
  return cfg
}

export function pickTag(name, tags) {
  if (!name || name.includes(':')) return name
  if (Array.isArray(tags)) {
    if (tags.includes(name)) return name
    if (tags.includes(`${name}:latest`)) return `${name}:latest`
  }
  return `${name}:latest`
}

export function resolveOpencodeModel(profile, tags) {
  const { ollamaModel, model } = profile.opencode
  if (!ollamaModel) return model
  return `ollama/${pickTag(ollamaModel, tags)}`
}

export function modelAvailability(tags, profiles = PROFILES) {
  const present = new Set(Array.isArray(tags) ? tags : [])
  return Object.entries(profiles).map(([name, p]) => {
    const model = p.opencode.ollamaModel || p.opencode.model
    const cloud = !p.opencode.ollamaModel || p.opencode.ollamaModel.includes(':')
    const installed = cloud
      ? null
      : present.has(model) || present.has(`${model}:latest`)
    return { profile: name, model, cloud, installed }
  })
}

export function matchProfile({ opencodeModel, clineModel }, profiles = PROFILES) {
  const norm = (value) => String(value || '').replace(/^ollama\//, '').replace(/:latest$/, '')
  if (!opencodeModel && !clineModel) return null

  const matches = Object.entries(profiles).filter(([, p]) => {
    const opencodeOk = !opencodeModel || norm(opencodeModel) === norm(p.opencode.model)
    const clineOk = !clineModel || norm(clineModel) === norm(p.cline.model)
    return opencodeOk && clineOk
  })
  return matches.length === 1 ? matches[0][0] : null
}

export function clinePatch(profile, ollamaUrl = OLLAMA_URL) {
  const patch = {
    welcomeViewCompleted: true,
    apiProvider: profile.cline.provider,
    apiModelId: profile.cline.model,
    planModeApiProvider: profile.cline.provider,
    actModeApiProvider: profile.cline.provider,
    planModeApiModelId: profile.cline.model,
    actModeApiModelId: profile.cline.model,
    planActSeparateModelsSetting: false,
  }

  if (profile.cline.provider === 'ollama') {
    Object.assign(patch, {
      ollamaModelId: profile.cline.model,
      ollamaBaseUrl: ollamaUrl,
      ollamaApiOptionsCtxNum: String(profile.cline.ctx),
      planModeOllamaModelId: profile.cline.model,
      actModeOllamaModelId: profile.cline.model,
    })
  }
  return patch
}

function stamp() {
  return new Date().toISOString().replace(/[:.]/g, '-').slice(0, 19)
}

async function ollamaTags() {
  try {
    const res = await fetch(`${OLLAMA_URL}/api/tags`, { signal: AbortSignal.timeout(3000) })
    if (!res.ok) return null
    const data = await res.json()
    return (data.models || []).map((m) => m.name)
  } catch {
    return null
  }
}

async function applyOpencode(profile, { root, dryRun }) {
  const file = path.join(root, 'opencode.json')
  const jsonc = path.join(root, 'opencode.jsonc')

  if (existsSync(jsonc) && !existsSync(file)) {
    warn(`✗ Найден ${path.basename(jsonc)}, а не opencode.json.`)
    warn('  Скрипт не правит .jsonc, чтобы не потерять комментарии.')
    warn('  Перенесите настройки в opencode.json или правьте вручную.')
    return false
  }

  let existing = null
  if (existsSync(file)) {
    try {
      existing = JSON.parse(readFileSync(file, 'utf8'))
    } catch (err) {
      warn(`✗ ${file} не разбирается как JSON: ${err.message}`)
      return false
    }
  }

  const tags = await ollamaTags()
  const model = resolveOpencodeModel(profile, tags)
  const check = validateModel(model, tags)
  if (!check.ok) warn(`⚠ ${check.message}`)

  const next = mergeOpencodeConfig({ ...profile, opencode: { ...profile.opencode, model } }, existing)
  const payload = `${JSON.stringify(next, null, 2)}\n`
  const before = existing ? readFileSync(file, 'utf8') : ''
  const changed = payload !== before

  log(`\nOpenCode: ${path.relative(root, file) || 'opencode.json'}`)
  log(`  model          = ${next.model}`)
  log(`  default_agent  = ${next.default_agent}`)

  if (!changed) {
    log('  изменений нет')
    return true
  }
  if (dryRun) {
    log('  --dry-run: не записано')
    return true
  }

  if (existsSync(file)) {
    const backup = `${file}.bak-${stamp()}`
    copyFileSync(file, backup)
    log(`  бэкап: ${path.basename(backup)}`)
  }
  writeFileSync(file, payload, 'utf8')
  log('  записано')
  return true
}

function validateModel(model, tags) {
  if (!model.startsWith('ollama/')) return { ok: true }
  const name = model.slice('ollama/'.length)
  if (name.endsWith('-cloud')) return { ok: true }
  if (!tags) {
    return { ok: false, message: `Ollama не отвечает на ${OLLAMA_URL} — модель не проверить` }
  }
  if (tags.includes(name) || tags.includes(`${name}:latest`)) return { ok: true }
  return {
    ok: false,
    message: `в Ollama нет модели «${name}». Проверьте: ollama list (или ollama create ...)`,
  }
}

function isVsCodeRunning() {
  if (process.platform !== 'win32') return false
  const images = ['Code.exe', 'Code - Insiders.exe']
  return images.some((image) => {
    try {
      const out = execFileSync('tasklist', ['/FI', `IMAGENAME eq ${image}`, '/NH'], {
        encoding: 'utf8',
        timeout: 5000,
      })
      return out.toLowerCase().includes(image.toLowerCase())
    } catch {
      return false
    }
  })
}

async function openClineDb(env = process.env, { readOnly = false } = {}) {
  const existing = clineDbCandidates(env).find((file) => existsSync(file))
  if (!existing) {
    warn('\nCline: не найден state.vscdb.')
    warn('  Похоже, VS Code с расширением Cline ещё не запускался на этой машине.')
    return null
  }
  try {
    const sqlite = await import('node:sqlite')
    // В новых Node класс называется Database, DatabaseSync — устаревший алиас.
    const Database = sqlite.Database || sqlite.DatabaseSync
    if (!Database) throw new Error('в node:sqlite нет класса Database')
    return { db: readOnly ? new Database(existing, { readOnly: true }) : new Database(existing), file: existing }
  } catch (err) {
    warn(`\nCline: node:sqlite недоступен (${err.message}).`)
    warn('  Нужен Node.js 22.13+ (на 22.5-22.12 — с флагом --experimental-sqlite).')
    return null
  }
}

function readCline(db) {
  const row = db.prepare('SELECT value FROM ItemTable WHERE key = ?').get(CLINE_KEY)
  if (row?.value == null) return {}
  const bytes = row.value
  const text = bytes instanceof Uint8Array ? Buffer.from(bytes).toString('utf8') : String(bytes)
  try {
    return JSON.parse(text)
  } catch {
    return {}
  }
}

async function applyCline(profile, { dryRun, force }) {
  const opened = await openClineDb()
  if (!opened) return false
  const { db, file } = opened

  try {
    const current = readCline(db)
    const patch = clinePatch(profile)
    const keys = Object.keys(patch)
    const changed = keys.some((k) => JSON.stringify(current[k]) !== JSON.stringify(patch[k]))

    log('\nCline:')
    for (const k of keys) {
      const before = current[k] === undefined ? '—' : String(current[k])
      const after = String(patch[k])
      log(`  ${before === after ? '=' : '~'} ${k}: ${before} → ${after}`)
    }

    if (!changed) {
      log('  изменений нет')
      return true
    }
    if (dryRun) {
      log('  --dry-run: не записано')
      return true
    }
    if (isVsCodeRunning() && !force) {
      warn('\n✗ VS Code запущен: настройки Cline не записаны.')
      warn('  Закройте VS Code и повторите (Cline читает состояние при старте).')
      warn('  Принудительно: добавьте --force (VS Code может перезаписать при выходе).')
      return false
    }

    db.exec('PRAGMA busy_timeout = 5000')
    const backup = `${file}.bak-${stamp()}`
    copyFileSync(file, backup)
    log(`  бэкап: ${path.basename(backup)}`)

    const next = { ...current, ...patch }
    const payload = JSON.stringify(next)
    db.exec('BEGIN IMMEDIATE')
    try {
      const row = db.prepare('SELECT value FROM ItemTable WHERE key = ?').get(CLINE_KEY)
      const value = row?.value instanceof Uint8Array ? Buffer.from(payload, 'utf8') : payload
      if (row) db.prepare('UPDATE ItemTable SET value = ? WHERE key = ?').run(value, CLINE_KEY)
      else db.prepare('INSERT INTO ItemTable (key, value) VALUES (?, ?)').run(CLINE_KEY, value)
      db.exec('COMMIT')
    } catch (err) {
      db.exec('ROLLBACK')
      throw err
    }
    log('  записано')
    return true
  } catch (err) {
    warn(`\n✗ Cline не обновлён: ${err.message}`)
    return false
  } finally {
    try {
      db.close()
    } catch {
      // база уже закрыта или недоступна — сообщать нечего
    }
  }
}

async function status(root) {
  log(`Проект: ${root}`)
  log(`Ollama:  ${OLLAMA_URL}`)

  const tags = await ollamaTags()
  if (tags) {
    log(`         локальных моделей: ${tags.length}`)
    log('         профили комплекта:')
    for (const row of modelAvailability(tags)) {
      const mark = row.cloud ? '☁' : row.installed ? '✅' : '❌'
      log(`           ${row.profile.padEnd(13)} ${mark} ${row.model}`)
    }
    log('         ❌ — модели нет: поставьте её через 02-Установить-локально.bat')
  } else {
    warn('         ✗ Ollama не отвечает (запустите Ollama или проверьте OLLAMA_URL)')
  }

  const opencodeFile = path.join(root, 'opencode.json')
  let opencodeModel = null
  if (existsSync(opencodeFile)) {
    try {
      const cfg = JSON.parse(readFileSync(opencodeFile, 'utf8'))
      opencodeModel = cfg.model || null
      log(`\nOpenCode: ${path.basename(opencodeFile)}`)
      log(`  model         = ${cfg.model || '—'}`)
      log(`  default_agent = ${cfg.default_agent || '—'}`)
    } catch {
      warn(`\nOpenCode: ${path.basename(opencodeFile)} не разбирается как JSON`)
    }
  } else {
    log('\nOpenCode: opencode.json нет (профиль ещё не применялся)')
  }

  const opened = await openClineDb()
  let clineModel = null
  if (opened) {
    try {

// Cline хранит base URL так, как его ввели; localhost и 127.0.0.1 — один адрес.
const normalizeBaseUrl = (value) =>
  String(value ?? '')
    .trim()
    .replace(/\/+$/, '')
    .replace(/^http:\/\/localhost(?=[:/]|$)/i, 'http://127.0.0.1')

export function verifyProfileState(profile, { cline = {}, opencode = null, tags = null } = {}) {
  const checks = []
  for (const [key, expected] of Object.entries(clinePatch(profile))) {
    const actual = cline[key]
    const compare = key === 'ollamaBaseUrl' ? normalizeBaseUrl : (v) => String(v ?? '')
    checks.push({
      scope: 'cline',
      name: key,
      expected: String(expected),
      actual: actual === undefined ? '—' : String(actual),
      ok: compare(actual) === compare(expected),
    })
  }

  if (opencode) {
    const model = resolveOpencodeModel(profile, tags)
    checks.push({
      scope: 'opencode',
      name: 'model',
      expected: model,
      actual: opencode.model || '—',
      ok: opencode.model === model,
    })
    checks.push({
      scope: 'opencode',
      name: 'default_agent',
      expected: profile.opencode.agent,
      actual: opencode.default_agent || '—',
      ok: opencode.default_agent === profile.opencode.agent,
    })
  }

  const failed = checks.filter((c) => !c.ok)
  return { ok: failed.length === 0, checks, failed }
}

// Обход дерева без служебных каталогов. Возвращает относительные пути
// в posix-форме и размеры; порядок детерминированный (сортировка в конце).
export function walkFiles(root, { maxFiles = 25000 } = {}) {
  const out = []
  const stack = [{ dir: root, rel: '' }]

  while (stack.length > 0 && out.length < maxFiles) {
    const { dir, rel } = stack.pop()
    let entries
    try {
      entries = readdirSync(dir, { withFileTypes: true })
    } catch {
      continue
    }
    entries.sort((a, b) => a.name.localeCompare(b.name))

    for (const entry of entries) {
      if (out.length >= maxFiles) break
      if (entry.isSymbolicLink && entry.isSymbolicLink()) continue

      const childRel = rel ? `${rel}/${entry.name}` : entry.name
      const full = path.join(dir, entry.name)

      if (entry.isDirectory()) {
        if (isNoiseDir(entry.name)) continue
        stack.push({ dir: full, rel: childRel })
      } else if (entry.isFile()) {
        let size = 0
        try {
          size = statSync(full).size
        } catch {
          // файл исчез между readdir и stat — пропускаем размер
        }
        out.push({ path: childRel, size })
      }
    }
  }

  out.sort((a, b) => a.path.localeCompare(b.path))
  return out
}

function runGit(root, args) {
  try {
    return execFileSync('git', ['-C', root, ...args], {
      encoding: 'utf8',
      stdio: ['ignore', 'pipe', 'ignore'],
      timeout: 5000,
    }).trim()
  } catch {
    return ''
  }
}

export function gitInfo(root) {
  const branch = runGit(root, ['rev-parse', '--abbrev-ref', 'HEAD'])
  const last = runGit(root, ['log', '-1', '--format=%h %s'])
  const dirty = runGit(root, ['status', '--porcelain'])
    .split('\n')
    .filter(Boolean).length
  return { branch, last, dirty }
}

export function readPackage(root) {
  const file = path.join(root, 'package.json')
  if (!existsSync(file)) return null
  try {
    const data = JSON.parse(readFileSync(file, 'utf8'))
    return data && typeof data === 'object' ? data : null
  } catch {

  }
}

export function detectEntryPoints(files, pkg) {
  const found = new Set()
  const add = (value) => {
    if (typeof value !== 'string') return
    const clean = toPosix(value).replace(/^\.\//, '')
    if (files.some((f) => f.path === clean)) found.add(clean)
  }

  if (pkg) {
    add(pkg.main)
    add(pkg.module)
    if (typeof pkg.bin === 'string') add(pkg.bin)
    else if (pkg.bin && typeof pkg.bin === 'object') Object.values(pkg.bin).forEach(add)
  }

  const re = /(^|\/)(?:src\/)?(server|app|main|index|cli)\.(ts|tsx|js|mjs|cjs|py|go|rs)$/i
  for (const file of files) {
    if (re.test(file.path)) found.add(file.path)
  }
  return [...found].sort().slice(0, 10)
}

function scanMarkers(root, files) {
  const counts = Object.fromEntries(MARKERS.map(([name]) => [name, 0]))
  let scanned = 0

  for (const file of files) {
    if (scanned >= MAX_SCAN_FILES) break
    if (file.size > MAX_SCAN_BYTES) continue
    const ext = path.posix.extname(file.path).toLowerCase()
    if (!CODE_EXT.has(ext)) continue
    // «any» — только в JS/TS: в Python/Go это слово встречается в тексте.
    const jsLike = ['.ts', '.tsx', '.js', '.jsx', '.mjs', '.cjs'].includes(ext)

    let text
    try {
      text = readFileSync(path.join(root, file.path), 'utf8')
    } catch {
      continue
    }
    scanned += 1

    for (const [name, re] of MARKERS) {
      if (name === 'any' && !jsLike) continue
      const matches = text.match(re)
      if (matches) counts[name] += matches.length
    }
  }

  return { ...counts, scanned }
}

function topDirs(files, limit = 15) {
  const counts = new Map()
  for (const file of files) {
    const slash = file.path.indexOf('/')
    const dir = slash === -1 ? '(корень)' : file.path.slice(0, slash)
    counts.set(dir, (counts.get(dir) || 0) + 1)
  }
  return [...counts.entries()]
    .sort((a, b) => b[1] - a[1] || a[0].localeCompare(b[0]))
    .slice(0, limit)
    .map(([dir, count]) => ({ dir, count }))
}

export function collectBrief(root, { limit = 20 } = {}) {
  const files = walkFiles(root)
  const pkg = readPackage(root)
  const git = gitInfo(root)
  const paths = files.map((file) => file.path)

  return {
    root: path.resolve(root),
    git,
    package: pkg
      ? {
          name: pkg.name || '',
          version: pkg.version || '',
          scripts: pkg.scripts || {},
          dependencies: Object.keys(pkg.dependencies || {}),
          devDependencies: Object.keys(pkg.devDependencies || {}),
        }
      : null,
    filesTotal: files.length,
    topDirs: topDirs(files),
    entry: detectEntryPoints(files, pkg),
    routes: paths.filter(isRouteFile).slice(0, limit),
    db: paths.filter(isDbFile).slice(0, limit),
    tests: paths.filter(isTestFile).slice(0, limit),
    testsTotal: paths.filter(isTestFile).length,
    configs: paths.filter(isConfigFile).slice(0, limit),
    largest: [...files]
      .sort((a, b) => b.size - a.size || a.path.localeCompare(b.path))
      .slice(0, limit)
      .map((file) => ({ path: file.path, bytes: file.size })),
    markers: scanMarkers(root, files),
  }
}

const kb = (bytes) => `${Math.max(1, Math.round(bytes / 1024))} КБ`

function list(title, items, limit) {
  if (items.length === 0) return [`${title}: —`]
  const shown = items.slice(0, limit)
  const tail = items.length > shown.length ? ` … ещё ${items.length - shown.length}` : ''
  return [`${title}: ${shown.join(', ')}${tail}`]
}

  if (brief.package) {
    const p = brief.package
    lines.push(`package: ${p.name || '—'} ${p.version || ''}`.trim())
    const scripts = Object.entries(p.scripts)

    lines.push(`scripts: ${scripts.length ? scripts.join(' · ') : '—'}`)
    if (p.dependencies.length) {

    if (p.devDependencies.length) {
      lines.push(

    }
  } else {
    lines.push('package: package.json нет')
  }

  lines.push(`files: ${brief.filesTotal} (без node_modules/dist/.git)`)
  lines.push(

export function parseArgs(argv, cwd = process.cwd()) {
  let root = cwd
  let out = ''
  let json = false
  let limit = 20

  for (let i = 0; i < argv.length; i += 1) {
    const arg = argv[i]
    if (arg === '--root') {
      const value = argv[i + 1]
      if (!value) throw new Error('--root требует путь')
      root = value
      i += 1
    } else if (arg.startsWith('--root=')) {
      const value = arg.slice('--root='.length)
      if (!value) throw new Error('--root требует путь')
      root = value
    } else if (arg === '--out') {
      const value = argv[i + 1]
      if (!value) throw new Error('--out требует путь')
      out = value
      i += 1
    } else if (arg.startsWith('--out=')) {
      out = arg.slice('--out='.length)
    } else if (arg === '--json') {
      json = true
    } else if (arg === '--limit') {
      const value = Number(argv[i + 1])
      if (!Number.isInteger(value) || value <= 0) throw new Error('--limit требует целое > 0')
      limit = value
      i += 1
    } else if (arg.startsWith('--limit=')) {
      const value = Number(arg.slice('--limit='.length))
      if (!Number.isInteger(value) || value <= 0) throw new Error('--limit требует целое > 0')
      limit = value
    } else if (arg === '--help' || arg === '-h') {
      return { help: true, root, out, json, limit }
    } else {
      throw new Error(`неизвестный аргумент: ${arg}`)
    }
  }

  return { help: false, root: path.resolve(root), out, json, limit }
}

import { existsSync, mkdirSync, readFileSync, writeFileSync } from 'node:fs'

// Причина провала — код, а не свободный текст: по нему видно, что чинить.
export const FAIL_REASONS = {
  hallucinated_file: 'выдуманный файл',
  hallucinated_line: 'выдуманная строка',
  gave_up_prematurely: 'сдался раньше',

const readJson = (file) => JSON.parse(readFileSync(file, 'utf8'))

export function loadTasks(file = DEFAULT_TASKS) {
  const tasks = readJson(file)
  if (!Array.isArray(tasks) || tasks.length === 0) throw new Error(`нет задач в ${file}`)
  const seen = new Set()
  for (const task of tasks) {
    if (!task || typeof task.id !== 'string' || !/^E-\d{2}$/u.test(task.id)) {
      throw new Error(`задача без ID формата E-NN в ${file}`)
    }
    if (seen.has(task.id)) throw new Error(`дубль ID ${task.id} в ${file}`)
    if (typeof task.prompt !== 'string' || task.prompt.trim().length < 20) {
      throw new Error(`у задачи ${task.id} нет промпта (prompt) в ${file}`)
    }
    seen.add(task.id)
  }
  return tasks
}

export function loadResults(file = DEFAULT_RESULTS) {
  if (!existsSync(file)) return { runs: [] }
  const data = readJson(file)
  if (!data || !Array.isArray(data.runs)) throw new Error(`битый файл результатов: ${file}`)
  return data
}

const isCount = (value) => Number.isInteger(value) && value >= 0

export function validateRun(run, tasks) {
  if (!run || !tasks.some((task) => task.id === run.id)) {
    throw new Error(`неизвестная задача: ${run?.id || '—'}`)
  }
  if (typeof run.pass !== 'boolean') throw new Error('нужен ровно один из флагов --pass / --fail')
  if (!isCount(run.steps)) throw new Error('--steps требует целое ≥ 0')
  if (!isCount(run.fabricated)) throw new Error('--fabricated требует целое ≥ 0')
  const reason = run.pass ? '' : String(run.reason || '').trim()
  if (reason && !Object.hasOwn(FAIL_REASONS, reason)) {
    throw new Error(`--reason: ${reason}; допустимо: ${Object.keys(FAIL_REASONS).join(', ')}`)
  }
  return { ...run, note: String(run.note || '').trim(), reason }
}

export function recordRun(data, run, tasks) {
  const clean = validateRun(run, tasks)
  const entry = { ...clean, at: run.at || new Date().toISOString() }
  return { ...data, runs: [...data.runs, entry] }
}

/** Последний прогон по каждой задаче; непройденные/незапущенные не считаются pass. */
export function scoreRuns(data, tasks) {
  const rows = tasks.map((task) => {
    const runs = data.runs.filter((run) => run.id === task.id)
    const last = runs.length > 0 ? runs[runs.length - 1] : null
    return {
      id: task.id,
      title: task.title,
      route: task.route,
      attempts: runs.length,
      pass: Boolean(last && last.pass),
      steps: last ? last.steps : null,
      fabricated: last ? last.fabricated : null,
      note: last ? last.note : '',
      reason: last ? last.reason || '' : '',
      at: last ? last.at : '',
    }
  })

  const attempted = rows.filter((row) => row.attempts > 0)
  const passed = rows.filter((row) => row.pass).length
  const fabricated = attempted.reduce((sum, row) => sum + (row.fabricated || 0), 0)
  const steps = attempted.filter((row) => row.steps !== null).map((row) => row.steps)
  const avgSteps = steps.length
    ? Math.round((steps.reduce((sum, value) => sum + value, 0) / steps.length) * 10) / 10
    : 0

  return {
    rows,
    total: rows.length,
    attempted: attempted.length,
    passed,
    fabricated,
    avgSteps,
    ready:
      passed >= THRESHOLDS.minPass &&
      fabricated <= THRESHOLDS.maxFabricated &&
      (attempted.length === 0 || avgSteps <= THRESHOLDS.maxAvgSteps),
  }
}

export function formatPrompt(task) {
  return [
    `# ${task.id} — ${task.title}`,
    `Маршрут: ${task.route}`,
    `Критерий pass: ${task.criterion}`,
    '',
    task.prompt,
    '',
    'Правила ответа:',
    '- цикл RESTATE → MAP → PLAN → ACT → VERIFY → REPORT;',
    '- не больше 5 прочитанных файлов и одного изменения;',

    '  и предложи облако (это тоже правильный ответ);',
    '- в конце — «Не проверено» и один следующий шаг.',
  ].join('\n')
}

export function formatScore(score) {
  const lines = ['ID    задача                          попыток  итог   шаги  выдумано']
  for (const row of score.rows) {
    const verdict = row.attempts === 0 ? '—' : row.pass ? 'PASS' : 'FAIL'
    lines.push(
      `${row.id.padEnd(5)} ${String(row.title).padEnd(31)} ${String(row.attempts).padStart(7)}  ` +
        `${verdict.padEnd(5)} ${String(row.steps ?? '—').padStart(4)}  ${String(row.fabricated ?? '—').padStart(9)}`,
    )
    if (row.note) lines.push(`      … ${row.note}`)
    if (!row.pass && row.reason) lines.push(`      ! ${row.reason} — ${FAIL_REASONS[row.reason] || row.reason}`)
  }
  lines.push('')
  lines.push(
    `Итог: ${score.passed}/${score.total} pass, пройдено задач ${score.attempted}/${score.total}, ` +
      `среднее шагов ${score.avgSteps}, выдумано ${score.fabricated}`,
  )
  lines.push(
    `Порог: pass ≥ ${THRESHOLDS.minPass}, выдумано ≤ ${THRESHOLDS.maxFabricated}, ` +
      `среднее шагов ≤ ${THRESHOLDS.maxAvgSteps} — ${score.ready ? 'готов ✓' : 'не готов ✗'}`,
  )
  return lines.join('\n')
}

export function parseArgs(argv) {
  const mode = argv[0]
  const out = { mode, flags: new Set(), values: {} }
  for (let i = 1; i < argv.length; i += 1) {
    const arg = argv[i]
    if (!arg.startsWith('--')) throw new Error(`неизвестный аргумент: ${arg}`)
    const key = arg.slice(2)
    if (['pass', 'fail', 'strict', 'help'].includes(key)) {
      out.flags.add(key)
      continue
    }
    const value = argv[i + 1]
    if (value === undefined || value.startsWith('--')) throw new Error(`--${key} требует значение`)
    out.values[key] = value
    i += 1
  }
  if (out.flags.has('pass') && out.flags.has('fail')) {
    throw new Error('--pass и --fail взаимоисключающие')
  }
  return out
}

function main(argv) {
  let args
  try {
    args = parseArgs(argv)
  } catch (err) {
    console.error(`✗ ${err.message}`)
    return 2
  }

  if (!args.mode || args.mode === '--help' || args.flags.has('help')) {
    console.log(usage())
    return args.mode ? 0 : 2
  }
  if (!['list', 'prompt', 'record', 'score'].includes(args.mode)) {
    console.error(`✗ неизвестная команда: ${args.mode}`)
    console.error(usage())
    return 2
  }

  const tasksFile = path.resolve(args.values.tasks || DEFAULT_TASKS)
  const resultsFile = path.resolve(args.values.results || DEFAULT_RESULTS)

  let tasks
  try {
    tasks = loadTasks(tasksFile)
  } catch (err) {
    console.error(`✗ ${err.message}`)
    return 2
  }

  if (args.mode === 'list') {
    for (const task of tasks) {
      console.log(`${task.id}\t${task.route.padEnd(5)}\t${task.title}`)
    }
    return 0
  }

  if (args.mode === 'prompt') {
    if (!args.values.id) {
      console.error('✗ --id обязателен (например --id E-01)')
      return 2
    }
    const task = tasks.find((item) => item.id === args.values.id)
    if (!task) {
      console.error(`✗ неизвестная задача: ${args.values.id}`)
      return 1
    }
    console.log(formatPrompt(task))
    return 0
  }

  let data
  try {
    data = loadResults(resultsFile)
  } catch (err) {
    console.error(`✗ ${err.message}`)
    return 2
  }

  if (args.mode === 'record') {
    try {
      const value = (key) => {
        const raw = args.values[key]
        if (raw === undefined) return 0
        const num = Number(raw)
        if (!Number.isInteger(num) || num < 0) throw new Error(`--${key} требует целое ≥ 0`)
        return num
      }
      const next = recordRun(
        data,
        {
          id: args.values.id,
          pass: args.flags.has('pass'),
          steps: value('steps'),
          fabricated: value('fabricated'),
          reason: args.values.reason || '',
          note: args.values.note || '',
        },
        tasks,
      )
      mkdirSync(path.dirname(resultsFile), { recursive: true })
      writeFileSync(resultsFile, `${JSON.stringify(next, null, 2)}\n`, 'utf8')
      const verdict = args.flags.has('pass') ? 'PASS' : 'FAIL'
      console.log(`✓ записано ${args.values.id}: ${verdict}`)
      return 0
    } catch (err) {
      console.error(`✗ ${err.message}`)
      return 1
    }
  }

  const score = scoreRuns(data, tasks)
  console.log(formatScore(score))
  if (args.flags.has('strict') && !score.ready) return 1
  return 0
}

export function parseArgs(argv = process.argv.slice(2)) {

function copyDecision({ source, target, force, immutable = false, memory = false, forceMemory = false }) {
  if (!existsSync(target)) {
    return { type: 'copy', source, target, reason: 'новый файл' }
  }
  if (!statSync(target).isFile()) {
    return { type: 'conflict', source, target, reason: 'цель существует, но это не файл' }
  }
  if (normalize(readFileSync(source, 'utf8')) === normalize(readFileSync(target, 'utf8'))) {
    return { type: 'skip', source, target, reason: 'уже совпадает' }
  }
  if (immutable) {
    return { type: 'skip', source, target, reason: 'существующий паспорт проекта сохранён' }
  }
  if (memory) {
    if (!forceMemory) {
      return { type: 'skip', source, target, reason: 'Memory Bank сохранён; нужен --force-memory' }
    }
    return { type: 'copy', source, target, reason: 'перезапись по --force-memory' }
  }
  if (!force) {
    return { type: 'skip', source, target, reason: 'отличается; нужен --force' }
  }
  return { type: 'copy', source, target, reason: 'перезапись по --force' }
}

  plan.actions.push(copyDecision({ source, target, ...options, ...extra }))
}

export function buildPlan(options) {
  const opts = { ...DEFAULT_OPTIONS, ...options }
  const root = path.resolve(opts.root)

  if (root === KIT_ROOT) {

  if (!existsSync(root) || !statSync(root).isDirectory()) {
    throw new UserError(`нет каталога проекта: ${root}`)
  }

  const plan = { root, actions: [], warnings: [] }
  const rulesPath = path.join(root, '.clinerules')

  if (existsSync(rulesPath)) {
    if (statSync(rulesPath).isDirectory()) {
      // уже каталог — работаем с ним
    } else if (opts.convertRules) {
      plan.actions.push({ type: 'convert-rules', source: rulesPath, target: rulesPath })
    } else {
      plan.actions.push({
        type: 'conflict',
        target: rulesPath,
        reason: 'это файл; workflows требуют каталог. Запустите с --convert-rules (файл сохранится как 00-project-rules.md)',
      })
      return plan
    }
  } else {
    addMkdir(plan, rulesPath)
  }

  const ruleFiles = [
    { source: path.join(RULES_DIR, '00-project.md'), name: '00-project.md', immutable: true },
    { source: path.join(TEMPLATES, 'agent-protocol.md'), name: '40-agent-cycle.md' },
    { source: path.join(TEMPLATES, 'few-shot.md'), name: '50-few-shot.md' },
  ]
  for (const rule of ruleFiles) {
    planFile(plan, opts, rule.source, path.join(rulesPath, rule.name), { immutable: rule.immutable })
  }

  if (!opts.noWorkflows) {
    const workflowsPath = path.join(rulesPath, 'workflows')
    addMkdir(plan, workflowsPath)
    for (const name of listMarkdown(WORKFLOWS_DIR)) {
      planFile(plan, opts, path.join(WORKFLOWS_DIR, name), path.join(workflowsPath, name))
    }
  }

  if (!opts.noMemory) {
    const memoryPath = path.join(root, 'cline_docs')
    addMkdir(plan, memoryPath)
    for (const name of listMarkdown(MEMORY_DIR)) {
      planFile(plan, opts, path.join(MEMORY_DIR, name), path.join(memoryPath, name), { memory: true })
    }
  }

  if (!opts.noDocs) {
    const docsPath = path.join(root, 'docs')
    addMkdir(plan, docsPath)
    planFile(plan, opts, path.join(DOCS_DIR, 'AGENTS-cline-setup.md'), path.join(docsPath, 'AGENTS-cline-setup.md'))
    planFile(plan, opts, path.join(TEMPLATES, 'verify-checklist.md'), path.join(docsPath, 'AGENTS-verify-checklist.md'))
  }

  return plan
}

export function applyPlan(plan, options = {}) {
  const opts = { ...DEFAULT_OPTIONS, ...options }
  const results = []

  for (const action of plan.actions) {
    if (action.type === 'skip' || action.type === 'conflict') {
      results.push({ ...action, applied: false })
      continue
    }
    if (opts.dryRun) {
      results.push({ ...action, applied: false, dryRun: true })
      continue
    }

    try {
      if (action.type === 'mkdir') {
        mkdirSync(action.target, { recursive: true })
        results.push({ ...action, applied: true })
      } else if (action.type === 'copy') {
        mkdirSync(path.dirname(action.target), { recursive: true })
        copyFileSync(action.source, action.target)
        results.push({ ...action, applied: true })
      } else if (action.type === 'convert-rules') {
        const backup = `${action.source}.kit-backup`
        const safeBackup = existsSync(backup) ? `${backup}.${Date.now()}` : backup
        renameSync(action.source, safeBackup)
        try {
          mkdirSync(action.target, { recursive: true })
          writeFileSync(path.join(action.target, '00-project-rules.md'), readFileSync(safeBackup))
        } catch (error) {
          // Не терять старые правила, если создать каталог или файл не удалось.
          try {
            if (!existsSync(action.source)) renameSync(safeBackup, action.source)
          } catch {
            // Если вернуть файл нельзя, бэкап остаётся рядом: путь виден в ошибке.
          }
          throw error
        }
        if (existsSync(safeBackup)) rmSync(safeBackup)
        results.push({ ...action, applied: true, movedTo: path.join(action.target, '00-project-rules.md') })
      } else {
        results.push({ ...action, applied: false, reason: `неизвестное действие: ${action.type}` })
      }
    } catch (error) {
      results.push({ ...action, applied: false, error: error.message })
    }
  }

  return results
}

export function installProject(options = {}) {
  const plan = buildPlan(options)
  const results = applyPlan(plan, options)
  const failed = results.filter((result) => result.type === 'conflict' || result.error)
  return { ...plan, results, ok: failed.length === 0 }
}

const relative = (root, target) => path.relative(root, target).replace(/\\/g, '/') || '.'

export function formatResult(result, root, dryRun = false) {
  const prefix = dryRun ? '[dry-run] ' : ''
  const target = result.target ? relative(root, result.target) : '—'
  if (result.type === 'mkdir') return `${prefix}[DIR ] ${target}`
  if (result.type === 'copy') return `${prefix}[COPY] ${target} — ${result.reason || ''}`.trimEnd()
  if (result.type === 'skip') return `${prefix}[SKIP] ${target} — ${result.reason || ''}`.trimEnd()
  if (result.type === 'conflict') return `${prefix}[CONFLICT] ${target} — ${result.reason || ''}`.trimEnd()
  if (result.type === 'convert-rules') {
    return `${prefix}[CONVERT] .clinerules -> .clinerules/00-project-rules.md`
  }
  return `${prefix}[?] ${target} — ${result.reason || result.error || result.type}`
}

export function runCli(argv = process.argv.slice(2), io = console) {
  try {
    const options = parseArgs(argv)
    if (options.help) {
      io.log(USAGE)
      return 0
    }

    const result = installProject(options)
    for (const row of result.results) io.log(formatResult(row, result.root, options.dryRun))

    const conflicts = result.results.filter((row) => row.type === 'conflict')
    const errors = result.results.filter((row) => row.error)
    const changed = result.results.filter(
      (row) => row.applied || (options.dryRun && (row.type === 'copy' || row.type === 'mkdir' || row.type === 'convert-rules')),
    )

    if (conflicts.length > 0) {
      io.error('\n✗ есть конфликты: установка неполная. Исправьте и повторите.')
      return 1
    }
    if (errors.length > 0) {
      io.error('\n✗ часть файлов не записана. Проверьте права и повторите.')
      return 1
    }

    const verb = options.dryRun ? 'будет создано/обновлено' : 'создано/обновлено'
    io.log(`\n✓ ${verb}: ${changed.length}; пропущено: ${result.results.length - changed.length}`)
    if (options.dryRun) io.log('  Это был --dry-run: файлы не изменены.')
    return 0
  } catch (error) {
    io.error(`✗ ${error.message}`)
    return error instanceof UserError ? 2 : 1
  }
}

const invokedDirectly =
  process.argv[1] && import.meta.url === pathToFileURL(path.resolve(process.argv[1])).href
if (invokedDirectly) {
  process.exitCode = runCli(process.argv.slice(2))
}
