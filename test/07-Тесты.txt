describe('E-07: формат «Файл/Строка» и сверка правила', () => {
  let dir
  afterEach(() => {
    if (dir) rmSync(dir, { recursive: true, force: true })
    dir = null
  })

  const make = (files) => {
    dir = mkdtempSync(path.join(tmpdir(), 'agent-e07-'))
    for (const [name, body] of Object.entries(files)) {
      const full = path.join(dir, name)
      mkdirSync(path.dirname(full), { recursive: true })
      writeFileSync(full, body, 'utf8')
    }
    return dir
  }

  const answer = (left, right) => `Нашёл дубль бизнес-правила.

Место 1:
**Файл:** \`${left.file}\`
**Строка:** ${left.line}
**Код:** \`${left.code}\`

Место 2:
**Файл:** \`${right.file}\`
**Строка:** ${right.line}
**Код:** \`${right.code}\`

Гроза: если менять лимит только в одном месте, поведение разойдётся.`

  const judge = (root, text) => {
    const check = checkCitations(root, citationsOf(text))
    return autoVerdict({ ...TASK, id: 'E-07' }, {
      answer: text,
      citations: check.checked,
      fabricated: check.fabricated,
      badLines: check.badLines,
      root,
    })
  }

  it('берёт номер строки из блока «Файл» + «Строка»', () => {
    const text = answer(
      { file: 'a.js', line: 3, code: 'export const MAX_LINES = 450' },
      { file: 'b.js', line: 2, code: 'const MAX_TEXT = 4000' },
    )
    const citations = citationsOf(text)
    expect(citations.every((item) => item.line === null)).toBe(true)
    const enriched = withDeclaredLines(text, citations)
    expect(enriched[0]).toMatchObject({ path: 'a.js', line: 3 })
    expect(enriched[1]).toMatchObject({ path: 'b.js', line: 2 })
  })

  it('не принимает две разные константы за одно правило', () => {
    const root = make({
      'a.js': 'one\ntwo\nexport const MAX_LINES = 450\n',
      'b.js': 'one\nconst MAX_TEXT = 4000\n',
    })
    const verdict = judge(root, answer(
      { file: 'a.js', line: 3, code: 'export const MAX_LINES = 450' },
      { file: 'b.js', line: 2, code: 'const MAX_TEXT = 4000' },
    ))
    expect(verdict.pass).toBe(false)
    expect(verdict.reason).toMatch(/разные правила/)
  })

  it('принимает одну константу в двух файлах', () => {
    const root = make({
      'a.js': 'one\ntwo\nconst MAX_TEXT = 4000\n',
      'b.js': 'one\nconst MAX_TEXT = 4000\n',
    })
    const verdict = judge(root, answer(
      { file: 'a.js', line: 3, code: 'const MAX_TEXT = 4000' },
      { file: 'b.js', line: 2, code: 'const MAX_TEXT = 4000' },
    ))
    expect(verdict.pass).toBe(true)
  })

  it('отдельная причина, если места названы без номеров строк', () => {
    const root = make({ 'a.js': 'const MAX_TEXT = 4000\n', 'b.js': 'const MAX_TEXT = 4000\n' })
    const check = checkCitations(root, citationsOf('Дубль в `a.js` и `b.js`. '.repeat(6)))
    const verdict = autoVerdict({ ...TASK, id: 'E-07' }, {
      answer: 'Дубль в `a.js` и `b.js`. '.repeat(6),
      citations: check.checked,
      fabricated: check.fabricated,
      badLines: check.badLines,
      root,
    })
    expect(verdict.pass).toBe(false)
    expect(verdict.reason).toMatch(/формат path:line/)
  })
})

  THRESHOLDS,
  formatPrompt,
  formatScore,
  loadResults,
  loadTasks,
  parseArgs,
  recordRun,
  scoreRuns,
  validateRun,
} from './agent-eval.mjs'

const SCRIPT = path.join(import.meta.dirname, 'agent-eval.mjs')
const TASKS = loadTasks()

const run = (id, pass, steps, fabricated = 0, note = '') => ({
  id,
  pass,
  steps,
  fabricated,
  note,
})

describe('задачи', () => {
  it('содержит 10 задач с уникальными ID', () => {
    expect(TASKS).toHaveLength(10)
    expect(new Set(TASKS.map((task) => task.id)).size).toBe(10)
    expect(TASKS[0].id).toBe('E-01')
    expect(TASKS[9].id).toBe('E-10')
  })

  it('у каждой задачи есть маршрут и критерий', () => {
    for (const task of TASKS) {
      expect(['local', 'cloud']).toContain(task.route)
      expect(task.criterion.length).toBeGreaterThan(20)
    }
  })

  it('у каждой задачи есть готовый промпт без заглушек', () => {
    for (const task of TASKS) {
      expect(task.prompt.length).toBeGreaterThan(60)
      expect(task.prompt).not.toMatch(/TODO|\.\.\.|<файл>|<ID>/u)
    }
  })

  it('formatPrompt печатает задачу, критерий и правила', () => {
    const text = formatPrompt(TASKS[0])
    expect(text).toContain('E-01')
    expect(text).toContain(TASKS[0].title)
    expect(text).toContain('Критерий pass:')
    expect(text).toContain('file:line')
    expect(text).toContain('Не проверено')
  })
})

describe('запись прогона', () => {
  it('проверяет ID, флаг и числа', () => {
    expect(() => validateRun(run('E-99', true, 1), TASKS)).toThrow(/неизвестная задача/)
    expect(() => validateRun({ id: 'E-01', steps: 1 }, TASKS)).toThrow(/--pass \/ --fail/)
    expect(() => validateRun(run('E-01', true, -1), TASKS)).toThrow(/--steps/)
    expect(() => validateRun(run('E-01', true, 1, -1), TASKS)).toThrow(/--fabricated/)
  })

  it('добавляет запись, не меняя исходные данные', () => {
    const data = { runs: [] }
    const next = recordRun(data, run('E-01', true, 4, 0, '  ок  '), TASKS)
    expect(data.runs).toHaveLength(0)
    expect(next.runs).toHaveLength(1)
    expect(next.runs[0].note).toBe('ок')
    expect(next.runs[0].at).toMatch(/^\d{4}-\d{2}-\d{2}T/)
  })
})

describe('счёт', () => {
  it('берёт последний прогон по каждой задаче', () => {
    const data = {
      runs: [
        run('E-01', false, 9, 2),
        run('E-01', true, 4, 0),
        run('E-02', true, 3),
      ],
    }
    const score = scoreRuns(data, TASKS)
    const e01 = score.rows.find((row) => row.id === 'E-01')
    expect(e01.pass).toBe(true)
    expect(e01.attempts).toBe(2)
    expect(e01.steps).toBe(4)
    expect(score.attempted).toBe(2)
    expect(score.passed).toBe(2)
    expect(score.fabricated).toBe(0)
  })

  it('считает выдуманное по последнему прогону', () => {
    const data = { runs: [run('E-01', true, 4), run('E-02', false, 9, 2)] }
    const score = scoreRuns(data, TASKS)
    expect(score.passed).toBe(1)
    expect(score.fabricated).toBe(2)
    expect(score.avgSteps).toBe(6.5)
  })

  it('готовность требует pass ≥ 7, 0 выдуманного и среднее ≤ 6', () => {
    const passing = { runs: TASKS.slice(0, 7).map((task) => run(task.id, true, 4)) }
    expect(scoreRuns(passing, TASKS).ready).toBe(true)

    const fabricated = { runs: TASKS.map((task) => run(task.id, true, 4, 1)) }
    expect(scoreRuns(fabricated, TASKS).ready).toBe(false)

    const slow = { runs: TASKS.map((task) => run(task.id, true, 7)) }
    expect(scoreRuns(slow, TASKS).ready).toBe(false)

    expect(THRESHOLDS.minPass).toBe(7)
  })

  it('пустой файл результатов не считается готовым', () => {
    const score = scoreRuns({ runs: [] }, TASKS)
    expect(score.attempted).toBe(0)
    expect(score.ready).toBe(false)
  })

  it('печатает таблицу и порог', () => {
    const text = formatScore(scoreRuns({ runs: [run('E-01', true, 4)] }, TASKS))
    expect(text).toContain('E-01')
    expect(text).toContain('PASS')
    expect(text).toContain('Порог:')
    expect(text).toContain('не готов')
  })
})

describe('аргументы', () => {
  it('понимает флаги и значения', () => {
    const args = parseArgs(['record', '--id', 'E-01', '--pass', '--steps', '4'])
    expect(args.mode).toBe('record')
    expect(args.values.id).toBe('E-01')
    expect(args.flags.has('pass')).toBe(true)
  })

  it('запрещает pass+fail одновременно', () => {
    expect(() => parseArgs(['record', '--id', 'E-01', '--pass', '--fail'])).toThrow(
      /взаимоисключающие/,
    )
  })

  it('падает на неизвестном аргументе', () => {
    expect(() => parseArgs(['list', 'лишнее'])).toThrow(/неизвестный/)
    expect(() => parseArgs(['record', '--id'])).toThrow(/--id/)
  })
})

describe('CLI', () => {
  let dir
  let results

  beforeEach(() => {
    dir = mkdtempSync(path.join(tmpdir(), 'eval-'))
    results = path.join(dir, 'results.json')
    writeFileSync(results, '{"runs":[]}\n', 'utf8')
  })

  afterEach(() => {
    rmSync(dir, { recursive: true, force: true })
  })

  const exec = (args) =>
    spawnSync(process.execPath, [SCRIPT, ...args], { encoding: 'utf8', timeout: 30000 })

  it('list печатает задачи', () => {
    const result = exec(['list'])
    expect(result.status).toBe(0)
    expect(result.stdout).toContain('E-01')
    expect(result.stdout).toContain('E-10')
  })

  it('prompt печатает готовый промпт, ошибки — 1/2', () => {
    const ok = exec(['prompt', '--id', 'E-01'])
    expect(ok.status).toBe(0)
    expect(ok.stdout).toContain('E-01')
    expect(ok.stdout).toContain('Критерий pass:')

    expect(exec(['prompt', '--id', 'E-99']).status).toBe(1)
    expect(exec(['prompt']).status).toBe(2)
  })

  it('record пишет результат, score его видит', () => {
    const record = exec([
      'record',
      '--id',
      'E-01',
      '--pass',
      '--steps',
      '4',
      '--fabricated',
      '0',
      '--note',
      'ок',
      '--results',
      results,
    ])
    expect(record.status).toBe(0)
    const data = loadResults(results)
    expect(data.runs).toHaveLength(1)
    expect(data.runs[0]).toMatchObject({ id: 'E-01', pass: true, steps: 4, note: 'ок' })

    const score = exec(['score', '--results', results])
    expect(score.status).toBe(0)
    expect(score.stdout).toContain('E-01')
    expect(score.stdout).toContain('1/10 pass')
  })

  it('record отдаёт 1 на неизвестной задаче и плохом числе', () => {
    const badId = exec(['record', '--id', 'E-99', '--pass', '--results', results])
    expect(badId.status).toBe(1)
    expect(badId.stderr).toMatch(/неизвестная задача/)

    const badSteps = exec([
      'record',
      '--id',
      'E-01',
      '--pass',
      '--steps',
      '-2',
      '--results',
      results,
    ])
    expect(badSteps.status).toBe(1)
    expect(badSteps.stderr).toMatch(/--steps/)
  })

  it('score --strict отдаёт 1, пока порог не достигнут', () => {
    const result = exec(['score', '--strict', '--results', results])
    expect(result.status).toBe(1)
    expect(result.stdout).toContain('не готов')
  })

  it('--help отдаёт 0, неизвестная команда — 2', () => {
    expect(exec(['--help']).status).toBe(0)
    expect(exec(['nope']).status).toBe(2)
  })
})

describe('битые файлы', () => {
  it('loadResults на отсутствующем файле возвращает пусто', () => {
    expect(loadResults(path.join(tmpdir(), 'no-such-eval-file.json'))).toEqual({ runs: [] })
  })

  it('loadTasks падает на дубле ID', () => {
    const dir = mkdtempSync(path.join(tmpdir(), 'eval-bad-'))
    const file = path.join(dir, 'tasks.json')
    try {
      writeFileSync(
        file,
        JSON.stringify([
          { id: 'E-01', title: 'a', route: 'local', criterion: 'x'.repeat(30), prompt: 'y'.repeat(80) },
          { id: 'E-01', title: 'b', route: 'local', criterion: 'x'.repeat(30), prompt: 'y'.repeat(80) },
        ]),
        'utf8',
      )
      expect(() => loadTasks(file)).toThrow(/дубль ID/)
    } finally {
      rmSync(dir, { recursive: true, force: true })
    }
  })

  it('не перезаписывает файл результатов при ошибке валидации', () => {
    const dir = mkdtempSync(path.join(tmpdir(), 'eval-keep-'))
    const file = path.join(dir, 'results.json')
    const before = '{"runs":[]}\n'
    try {
      writeFileSync(file, before, 'utf8')
      const result = spawnSync(
        process.execPath,
        [SCRIPT, 'record', '--id', 'E-99', '--pass', '--results', file],
        { encoding: 'utf8', timeout: 30000 },
      )
      expect(result.status).toBe(1)
      expect(readFileSync(file, 'utf8')).toBe(before)
    } finally {
      rmSync(dir, { recursive: true, force: true })
    }
  })
})

import { afterEach, describe, expect, it } from 'vitest'
import { existsSync, mkdtempSync, readFileSync, rmSync, writeFileSync } from 'node:fs'
import { tmpdir } from 'node:os'
import path from 'node:path'
import {
  KIT_ROOT,
  UserError,
  applyPlan,
  installProject,
  parseArgs,
  runCli,
} from './init-project.mjs'

const tempDirs = []

function tempProject() {
  const dir = mkdtempSync(path.join(tmpdir(), 'kit-init-'))
  tempDirs.push(dir)
  return dir
}

const read = (root, name) => readFileSync(path.join(root, name), 'utf8')
const write = (root, name, text) => writeFileSync(path.join(root, name), text, 'utf8')

afterEach(() => {
  while (tempDirs.length > 0) rmSync(tempDirs.pop(), { recursive: true, force: true })
})

describe('init-project: аргументы', () => {
  it('разбирает --root и флаги', () => {
    const options = parseArgs(['--root', 'C:\\project', '--dry-run', '--force', '--convert-rules'])
    expect(options.root).toBe(path.resolve('C:\\project'))
    expect(options.dryRun).toBe(true)
    expect(options.force).toBe(true)
    expect(options.convertRules).toBe(true)
  })

  it('падает на неизвестном флаге и пустом --root', () => {
    expect(() => parseArgs(['--nope'])).toThrow(/неизвестный аргумент/u)
    expect(() => parseArgs(['--root'])).toThrow(/--root требует/u)
  })

  it('запрещает установку в сам комплект', () => {
    expect(() => installProject({ root: KIT_ROOT, dryRun: true })).toThrow(UserError)
  })
})

describe('init-project: установка', () => {
  it('создаёт правила, workflows, память и docs', () => {
    const root = tempProject()
    const result = installProject({ root })

    expect(result.ok).toBe(true)
    expect(existsSync(path.join(root, '.clinerules', '00-project.md'))).toBe(true)
    expect(existsSync(path.join(root, '.clinerules', '40-agent-cycle.md'))).toBe(true)
    expect(existsSync(path.join(root, '.clinerules', '50-few-shot.md'))).toBe(true)
    expect(existsSync(path.join(root, '.clinerules', 'workflows', 'deep-planning.md'))).toBe(true)
    expect(existsSync(path.join(root, 'cline_docs', 'activeContext.md'))).toBe(true)
    expect(existsSync(path.join(root, 'docs', 'AGENTS-cline-setup.md'))).toBe(true)
    expect(existsSync(path.join(root, 'docs', 'AGENTS-verify-checklist.md'))).toBe(true)
    expect(read(root, '.clinerules/40-agent-cycle.md')).toMatch(/RESTATE/u)
    expect(read(root, 'cline_docs/activeContext.md')).toMatch(/Текущая задача/u)
  })

  it('повторный запуск ничего не перезаписывает', () => {
    const root = tempProject()
    installProject({ root })
    const first = read(root, '.clinerules/40-agent-cycle.md')
    const second = installProject({ root })

    expect(second.ok).toBe(true)
    expect(second.results.filter((row) => row.type === 'copy' && row.applied)).toEqual([])
    expect(read(root, '.clinerules/40-agent-cycle.md')).toBe(first)
  })

  it('--dry-run не создаёт файлов', () => {
    const root = tempProject()
    const result = installProject({ root, dryRun: true })

    expect(result.ok).toBe(true)
    expect(result.results.some((row) => row.type === 'copy' && row.dryRun)).toBe(true)
    expect(existsSync(path.join(root, '.clinerules'))).toBe(false)
    expect(existsSync(path.join(root, 'cline_docs'))).toBe(false)
  })

  it('--no-workflows и --no-memory отключают части', () => {
    const root = tempProject()
    installProject({ root, noWorkflows: true, noMemory: true })

    expect(existsSync(path.join(root, '.clinerules', '40-agent-cycle.md'))).toBe(true)
    expect(existsSync(path.join(root, '.clinerules', 'workflows'))).toBe(false)
    expect(existsSync(path.join(root, 'cline_docs'))).toBe(false)
  })
})

describe('init-project: существующие файлы', () => {
  it('не перезаписывает правила без --force', () => {
    const root = tempProject()
    installProject({ root })
    write(root, '.clinerules/40-agent-cycle.md', 'МОИ ПРАВИЛА\n')

    const result = installProject({ root })
    expect(result.ok).toBe(true)
    expect(read(root, '.clinerules/40-agent-cycle.md')).toBe('МОИ ПРАВИЛА\n')
  })

  it('перезаписывает правила с --force', () => {
    const root = tempProject()
    installProject({ root })
    write(root, '.clinerules/40-agent-cycle.md', 'МОИ ПРАВИЛА\n')

    installProject({ root, force: true })
    expect(read(root, '.clinerules/40-agent-cycle.md')).toMatch(/RESTATE/u)
  })

  it('не перезаписывает паспорт проекта даже с --force', () => {
    const root = tempProject()
    installProject({ root })
    write(root, '.clinerules/00-project.md', 'МОЙ ПАСПОРТ\n')

    installProject({ root, force: true })
    expect(read(root, '.clinerules/00-project.md')).toBe('МОЙ ПАСПОРТ\n')
  })

  it('сохраняет Memory Bank без --force-memory', () => {
    const root = tempProject()
    installProject({ root })
    write(root, 'cline_docs/activeContext.md', 'МОЯ ПАМЯТЬ\n')

    installProject({ root, force: true })
    expect(read(root, 'cline_docs/activeContext.md')).toBe('МОЯ ПАМЯТЬ\n')
  })

  it('перезаписывает Memory Bank с --force-memory', () => {
    const root = tempProject()
    installProject({ root })
    write(root, 'cline_docs/activeContext.md', 'МОЯ ПАМЯТЬ\n')

    installProject({ root, forceMemory: true })
    expect(read(root, 'cline_docs/activeContext.md')).toMatch(/Текущая задача/u)
  })
})

describe('init-project: .clinerules-файл', () => {
  it('без --convert-rules возвращает конфликт и ничего не ломает', () => {
    const root = tempProject()
    write(root, '.clinerules', 'СТАРЫЕ ПРАВИЛА\n')

    const result = installProject({ root })
    expect(result.ok).toBe(false)
    expect(result.results.some((row) => row.type === 'conflict')).toBe(true)
    expect(read(root, '.clinerules')).toBe('СТАРЫЕ ПРАВИЛА\n')
    expect(existsSync(path.join(root, '.clinerules', '40-agent-cycle.md'))).toBe(false)
  })

  it('с --convert-rules сохраняет старый файл как 00-project-rules.md', () => {
    const root = tempProject()
    write(root, '.clinerules', 'СТАРЫЕ ПРАВИЛА\n')

    const result = installProject({ root, convertRules: true })
    expect(result.ok).toBe(true)
    expect(existsSync(path.join(root, '.clinerules', '00-project-rules.md'))).toBe(true)
    expect(read(root, '.clinerules/00-project-rules.md')).toBe('СТАРЫЕ ПРАВИЛА\n')
    expect(existsSync(path.join(root, '.clinerules', '40-agent-cycle.md'))).toBe(true)
    expect(existsSync(path.join(root, '.clinerules'))).toBe(true)
  })

  it('возвращает старый файл, если конвертация не удалась', () => {
    const root = tempProject()
    const source = path.join(root, '.clinerules')
    const target = path.join(root, 'blocked')
    writeFileSync(source, 'СТАРЫЕ ПРАВИЛА\n', 'utf8')
    writeFileSync(target, 'занято\n', 'utf8')

    const results = applyPlan({ root, actions: [{ type: 'convert-rules', source, target }] })
    expect(results[0].error).toBeTruthy()
    expect(readFileSync(source, 'utf8')).toBe('СТАРЫЕ ПРАВИЛА\n')
    expect(existsSync(`${source}.kit-backup`)).toBe(false)
  })
})

  it('возвращает 1 на конфликт .clinerules-файла', () => {
    const root = tempProject()

    expect(code).toBe(1)

  })
})
